    const questionsData = {
      math: {
        1: Array.from({ length: 25 }, (_, index) => ({
          question: `Math Question ${index + 1}`,
          options: [`Option A`, `Option B`, `Option C`, `Option D`],
          answer: `Option A`, // Placeholder answers, modify as needed
          difficulty: index < 10 ? 'easy' : (index < 20 ? 'medium' : 'hard'),
        })),
        2: Array.from({ length: 25 }, (_, index) => ({
          question: `Math Question ${index + 1}`,
          options: [`Option A`, `Option B`, `Option C`, `Option D`],
          answer: `Option A`, // Placeholder answers, modify as needed
          difficulty: index < 10 ? 'easy' : (index < 20 ? 'medium' : 'hard'),
        })),
      },
      science: {
        1: Array.from({ length: 25 }, (_, index) => ({
          question: `Science Question ${index + 1}`,
          options: [`Option A`, `Option B`, `Option C`, `Option D`],
          answer: `Option A`, // Placeholder answers, modify as needed
          difficulty: index < 10 ? 'easy' : (index < 20 ? 'medium' : 'hard'),
        })),
      },
    };

    let currentQuestionIndex = 0;
    let score = 0;
    let gameCode = "";
    let currentSubject = "math";
    let currentGrade = 1;
    let playerName = "";
    let playerAvatar = "";
    let startTime = 0;
    let questionTimeLimit = 30; // seconds to answer each question

    let players = [];
    let currentPlayerIndex = 0;

    function displayQuestion() {
      const questions = questionsData[currentSubject][currentGrade];
      const question = questions[currentQuestionIndex];
      document.getElementById("question").innerText = question.question;
      const optionsContainer = document.getElementById("options");
      optionsContainer.innerHTML = "";

      question.options.forEach(option => {
        const button = document.createElement("button");
        button.innerText = option;
        button.onclick = () => checkAnswer(option);
        optionsContainer.appendChild(button);
      });
      startTime = Date.now(); // Start the timer for this question
      document.getElementById("question-container").style.display = "block";
      setTimeout(() => checkAnswer(null), questionTimeLimit * 1000); // Automatically check answer after time limit
    }

    function checkAnswer(selectedAnswer) {
      const question = questionsData[currentSubject][currentGrade][currentQuestionIndex];
      const timeTaken = Math.max(1, Math.floor((Date.now() - startTime) / 1000)); // Time taken in seconds
      let points = 0;

      if (selectedAnswer === question.answer) {
        points = getPoints(question.difficulty, timeTaken);
        score += points;
        document.getElementById("scoreValue").innerText = score;
      }

      document.getElementById("nextButton").style.display = "block";
    }

    function getPoints(difficulty, timeTaken) {
      let basePoints = difficulty === 'hard' ? 20 : difficulty === 'medium' ? 15 : 10;
      let timeBonus = Math.max(1, 30 - timeTaken); // 30 is max time, gives bonus for faster answers
      return basePoints + timeBonus;
    }

    function nextQuestion() {
      currentQuestionIndex++;
      if (currentQuestionIndex < questionsData[currentSubject][currentGrade].length) {
        displayQuestion();
        document.getElementById("nextButton").style.display = "none";
      } else {
        alert("Game Over! Your final score is " + score);
      }
    }

    // Functionality for lobby creation, joining, and setting up avatars remains the same...

      document.getElementById("createLobbyButton").style.display = "inline-block";
      document.getElementById("startGameButton").style.display = "none"; // Initially hide the start button
      document.getElementById("playerSetup").style.display = "block";
    };
  </script>
